cmake_minimum_required(VERSION 3.28)
project(strife_stream LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This capture prototype currently targets Windows x64")
endif()
set(OBS_SOURCE "" CACHE PATH "Pinned built OBS checkout")
set(OBS_DEPS "${OBS_SOURCE}/.deps/obs-deps-2026-08-26-x64")
set(OBS_BUILD "${OBS_SOURCE}/build_x64")
add_executable(strife-stream main.cpp)
add_library(strife-webrtc MODULE
  "${WHIP_SOURCE}/obs-webrtc.cpp" "${WHIP_SOURCE}/whip-output.cpp" "${WHIP_SOURCE}/whip-service.cpp")
add_library(strife-wasapi MODULE
  "${WASAPI_SOURCE}/win-wasapi.cpp" "${WASAPI_SOURCE}/wasapi-notify.cpp"
  "${WASAPI_SOURCE}/enum-wasapi.cpp" "${WASAPI_SOURCE}/plugin-main.cpp")
foreach(target strife-stream strife-webrtc strife-wasapi)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_options(${target} PRIVATE /utf-8)
  target_compile_definitions(${target} PRIVATE HAVE_OBSCONFIG_H RTC_ENABLE_MEDIA=1 RTC_ENABLE_WEBSOCKET=1)
  target_include_directories(${target} PRIVATE "${OBS_SOURCE}/libobs" "${OBS_BUILD}/config" "${OBS_BUILD}/libobs" "${OBS_DEPS}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
  target_link_libraries(${target} PRIVATE "${OBS_BUILD}/libobs/$<CONFIG>/obs.lib")
endforeach()
target_link_libraries(strife-stream PRIVATE ole32)
target_compile_definitions(strife-stream PRIVATE NOMINMAX)
target_link_libraries(strife-webrtc PRIVATE "${OBS_DEPS}/lib/datachannel.lib" "${OBS_DEPS}/lib/libcurl_imp.lib")
set_target_properties(strife-webrtc PROPERTIES OUTPUT_NAME obs-webrtc)
target_include_directories(strife-wasapi PRIVATE "${OBS_SOURCE}/deps/w32-pthreads")
target_compile_definitions(strife-wasapi PRIVATE UNICODE _UNICODE)
target_link_libraries(strife-wasapi PRIVATE Avrt "${OBS_BUILD}/deps/w32-pthreads/$<CONFIG>/w32-pthreads.lib")
set_target_properties(strife-wasapi PROPERTIES OUTPUT_NAME win-wasapi)
